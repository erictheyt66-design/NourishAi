        // --- ROUTER ---
        function router() {
            const hash = window.location.hash || '#/';
            const app = document.getElementById('app');
            window.scrollTo(0, 0);

            if (hash === '#/' || hash === '') {
                renderHome(app);
            } else if (hash === '#/recipes') {
                renderRecipes(app);
            } else if (hash.startsWith('#/recipes/')) {
                const recipeId = hash.replace('#/recipes/', '');
                renderRecipeDetail(app, recipeId);
            } else if (hash === '#/chat') {
                renderChat(app);
            } else if (hash === '#/analyze') {
                renderAnalyze(app);
            } else if (hash === '#/planner') {
                renderPlanner(app);
            } else if (hash === '#/bmi') {
                renderBmi(app);
            } else {
                app.innerHTML = `<div class="max-w-7xl mx-auto px-4 py-24 text-center"><h1 class="font-serif text-3xl font-bold">Page not found</h1><p class="mt-4 text-olive-600">The page you are looking for does not exist.</p><a href="#/" class="mt-6 inline-block bg-olive-700 text-white px-6 py-2.5 rounded-xl">Return Home</a></div>`;
            }
            lucide.createIcons();
            updateActiveNav(hash);
        }

        function updateActiveNav(hash) {
            document.querySelectorAll('.nav-link').forEach(link => {
                if (link.getAttribute('href') === hash || (hash.startsWith('#/recipes') && link.getAttribute('href') === '#/recipes')) {
                    link.classList.add('bg-white', 'shadow-xs', 'text-olive-900');
                    link.classList.remove('text-olive-800');
                } else {
                    link.classList.remove('bg-white', 'shadow-xs', 'text-olive-900');
                    link.classList.add('text-olive-800');
                }
            });
        }

        window.addEventListener('hashchange', router);
        window.addEventListener('load', router);

        function renderRecipeCards(list) {
            if (list.length === 0) {
                return `<div class="col-span-3 text-center py-16 text-olive-600">No matching recipes found. Try another search term or filter!</div>`;
            }
            return list.map(recipe => `
                <a href="#/recipes/${recipe.id}" class="bg-white rounded-2xl overflow-hidden border border-olive-200 hover:shadow-lg transition flex flex-col group">
                    <div class="relative h-52 overflow-hidden bg-olive-100">
                        <img src="${recipe.image}" alt="${recipe.title}" class="w-full h-full object-cover group-hover:scale-105 transition duration-500">
                        <div class="absolute top-3 left-3 flex flex-wrap gap-1.5">
                            <span class="bg-olive-900/80 backdrop-blur-xs text-white text-[10px] font-mono px-2.5 py-1 rounded-full">${recipe.category}</span>
                        </div>
                    </div>
                    <div class="p-6 flex flex-col flex-grow justify-between space-y-4">
                        <div class="space-y-2">
                            <h3 class="font-serif text-xl font-bold text-olive-900 group-hover:text-olive-700 transition">${recipe.title}</h3>
                            <p class="text-xs text-olive-700 line-clamp-2">${recipe.subtitle}</p>
                            <div class="flex flex-wrap gap-1 pt-1">
                                ${recipe.tags.map(t => `<span class="text-[10px] font-mono bg-olive-50 text-olive-700 px-2 py-0.5 rounded-md border border-olive-200/50">${t}</span>`).join('')}
                            </div>

                            <span class="flex items-center gap-1"><i data-lucide="flame" class="w-3.5 h-3.5 text-amber-600"></i> ${recipe.calories} kcal</span>
                        </div>
                    </div>
                </a>
            `).join('');
        }

        window.filterRecipes = function() {
            const query = document.getElementById('recipe-search').value.toLowerCase();
            const filtered = RECIPES.filter(r => r.title.toLowerCase().includes(query) || r.subtitle.toLowerCase().includes(query) || r.category.toLowerCase().includes(query) || r.tags.some(t => t.toLowerCase().includes(query)));
            document.getElementById('recipes-grid').innerHTML = renderRecipeCards(filtered);
            lucide.createIcons();
        };

        window.filterCategory = function(cat) {
            document.querySelectorAll('.filter-btn').forEach(btn => {
                btn.classList.remove('bg-olive-700', 'text-white');
                btn.classList.add('bg-olive-100', 'text-olive-800');
            });
            event.target.classList.add('bg-olive-700', 'text-white');
            event.target.classList.remove('bg-olive-100', 'text-olive-800');

            if (cat === 'All') {
                document.getElementById('recipes-grid').innerHTML = renderRecipeCards(RECIPES);
            } else {
                const filtered = RECIPES.filter(r => r.category === cat || r.category2 === cat);
                document.getElementById('recipes-grid').innerHTML = renderRecipeCards(filtered);
            }
            lucide.createIcons();
        };

        window.handleChatSubmit = async function(e) {
            e.preventDefault();
            const inputEl = document.getElementById('chat-input');
            const text = inputEl.value.trim();
            if (!text) return;

            chatState.messages.push({ role: 'user', content: text });
            inputEl.value = '';
            renderChat(document.getElementById('app'));

            const msgContainer = document.getElementById('chat-messages');
            msgContainer.scrollTop = msgContainer.scrollHeight;

            // Add typing indicator
            const typingId = 'typing-' + Date.now();
            chatState.messages.push({ role: 'assistant', content: 'Thinking...', id: typingId });
            renderChat(document.getElementById('app'));
            msgContainer.scrollTop = msgContainer.scrollHeight;

            // Simulated AI Intelligent Response grounded in NourishAI recipes
            setTimeout(() => {
                chatState.messages = chatState.messages.filter(m => m.id !== typingId);
                
                let reply = `Based on your goal of **${chatState.goal}** and **${chatState.diet}** preference, here is my recommendation:\n\n`;
                const queryLower = text.toLowerCase();
                
                if (queryLower.includes('blood pressure') || queryLower.includes('dash') || queryLower.includes('sodium')) {
                    reply += `For blood pressure management, aiming for lower sodium (<400mg per meal) and high potassium is key. I recommend checking out our **Ginger lentil power bowl** (${window.formatPrice(1.85)}/serving, 680mg potassium) or our **Provençal rosemary chickpea stew**.\n\nWould you like the full ingredient list for either?`;
                } else if (queryLower.includes('cheap') || queryLower.includes('budget') || queryLower.includes('cost') || queryLower.includes('affordable')) {
                    reply += `On a ${chatState.budget} budget, lentils, beans, oats, and seasonal greens are your best friends. Our **Ginger lentil power bowl** costs only **${window.formatPrice(1.85)}** per serving and provides 18g of plant protein!`;
                } else if (queryLower.includes('sweet') || queryLower.includes('breakfast') || queryLower.includes('oat')) {
                    reply += `For a healthy no-cook breakfast, try the **Overnight oat & berry jar** (${window.formatPrice(2.10)}/serving). It features oats, Greek yogurt, chia seeds, and berries for steady energy and stable mood.`;
                } else {
                    reply += `That is a wonderful question! To support your **${chatState.goal}**, focus on whole foods, adequate fiber, and balanced macronutrients. You can explore our curated recipes under the Recipes tab or check out our meal planner to build a full daily schedule!`;
                }

                chatState.messages.push({ role: 'assistant', content: reply });
                renderChat(document.getElementById('app'));
                msgContainer.scrollTop = msgContainer.scrollHeight;
            }, 1000);
        };

        window.calculateBmi = function() {
            const w = parseFloat(document.getElementById('bmi-weight').value);
            const h = parseFloat(document.getElementById('bmi-height').value) / 100;
            if (!w || !h || h <= 0) return;
            const bmi = (w / (h * h)).toFixed(1);
            document.getElementById('bmi-val').innerText = bmi;
            
            let status = 'Normal weight range';
            if (bmi < 18.5) status = 'Underweight range';
            else if (bmi >= 25) status = 'Overweight range';
            document.getElementById('bmi-status').innerText = status;
            document.getElementById('bmi-cal').innerText = `~${Math.round(w * 31)} kcal`;
            document.getElementById('bmi-protein').innerText = `~${Math.round(w * 1.2)}g`;
        };
